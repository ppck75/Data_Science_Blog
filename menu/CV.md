  // TOC 클릭 시, 실제 스크롤 컨테이너(#contents)에서 정확히 이동하도록 강제
  (function(){
    function getScrollContainer(){
      return document.getElementById("contents") || document.scrollingElement || document.documentElement;
    }

    document.addEventListener("click", function(e){
      var a = e.target.closest(".cv-toc a[href^='#']");
      if(!a) return;

      e.preventDefault();
      var id = a.getAttribute("href");
      var target = document.querySelector(id);
      if(!target) return;

      // #contents가 스크롤 컨테이너인 경우를 고려
      var sc = getScrollContainer();
      var topOffset = 16;

      // scrollIntoView가 안 먹는 레이아웃 대비: 직접 위치 계산
      if (sc === document.documentElement || sc === document.body || sc === document.scrollingElement) {
        target.scrollIntoView({behavior: "smooth", block: "start"});
      } else {
        var y = target.getBoundingClientRect().top - sc.getBoundingClientRect().top + sc.scrollTop - topOffset;
        sc.scrollTo({top: y, behavior: "smooth"});
      }
    }, true);
  })();
